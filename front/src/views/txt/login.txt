<template>
  <div class="m-0 p-0 absolute inset-0 bg-cover bg-center overflow-hidden" :style="backgroundStyle">
    <div class="absolute inset-0 bg-black bg-opacity-40"></div>

    <div class="relative h-full flex items-center justify-end p-8">
      <div class="w-full md:w-1/3 max-w-md bg-white bg-opacity-10 backdrop-blur-lg border border-white border-opacity-20 rounded-xl p-8 shadow-2xl">
        <div class="text-center mb-10">
          <div class="text-3xl font-bold text-white mb-2">同济大学智能化考试命题查重平台</div>
        </div>

        <form class="space-y-6" @submit.prevent="handleLogin">
          <div>
            <label for="employeeId" class="block text-sm font-medium text-white mb-2">工号</label>
            <input
              id="employeeId"
              v-model="employeeId"
              type="text"
              placeholder="请输入工�?
              class="w-full px-4 py-3 bg-white bg-opacity-10 border border-white border-opacity-20 rounded-lg text-white placeholder-gray-300 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div>
            <label for="password" class="block text-sm font-medium text-white mb-2">密码</label>
            <input
              id="password"
              v-model="password"
              type="password"
              placeholder="请输入密�?
              class="w-full px-4 py-3 bg-white bg-opacity-10 border border-white border-opacity-20 rounded-lg text-white placeholder-gray-300 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div>
            <button
              type="submit"
              :disabled="loading"
              class="w-full py-3 px-4 bg-blue-600 hover:bg-blue-700 text-white font-semibold rounded-lg shadow-lg transition duration-300 ease-in-out focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-blue-500 disabled:opacity-50"
            >
              {{ loading ? '登录�?..' : '登录' }}
            </button>
          </div>

          <div class="text-center text-sm text-gray-400 mt-6">
            <a href="#" class="text-white hover:text-blue-300">忘记密码�?/a>
          </div>

          <!-- 错误提示 -->
          <div v-if="errorMsg" class="text-red-300 text-sm text-center bg-red-900 bg-opacity-50 py-2 rounded">
            {{ errorMsg }}
          </div>
        </form>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue';
import { useRouter } from 'vue-router';

const employeeId = ref('');
const password = ref('');
const loading = ref(false);
const errorMsg = ref('');

const router = useRouter();

const backgroundStyle = {
  'background-image': 'url(https://images.unsplash.com/photo-1519681393784-d120267933ba?q=80&w=1000)'
};

const handleLogin = async () => {
  if (!employeeId.value || !password.value) {
    errorMsg.value = '请输入工号和密码';
    return;
  }

  loading.value = true;
  errorMsg.value = '';

  try {
    // 调用后端认证接口
    const user = await window.electronAPI.user.authenticate(employeeId.value, password.value);
    
    if (!user) {
      errorMsg.value = '工号或密码错�?;
      return;
    }

    // 保存用户信息�?localStorage
    localStorage.setItem('currentUser', JSON.stringify(user));
    
    // 登录成功，跳转到主菜单页（或你希望的目标页面�?
    router.push('/menu');
  } catch (err) {
    console.error('登录异常:', err);
    errorMsg.value = '登录失败，请稍后重试';
  } finally {
    loading.value = false;
  }
};
</script>

<style scoped>
input:focus {
  box-shadow: 0 0 0 2px rgba(59, 130, 246, 0.5);
}
</style>
